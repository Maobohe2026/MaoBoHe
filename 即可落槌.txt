    NumFeatures(8) = "4FA2FF|-4|3|4FA2FF,5|3|4FA2FF,0|4|-4FA2FF,-7|8|-4FA2FF,0|8|4FA2FF,7|8|-4FA2FF,0|12|-4FA2FF,-5|13|4FA2FF,6|14|4FA2FF,3|16|4FA2FF,0|17|4FA2FF,8|17|-4FA2FF,-6|18|-4FA2FF"

    Result = ""
    CurrentX = x1 ' 记录当前查找的起始X坐标
    Do
        MinX = x2 + 1 ' 初始化为区域最右侧之外，用于寻找最小的X坐标
        MinNum = -1   ' 记录最靠左的数字是几
        ' 遍历0-9的特征，在当前行寻找“最靠左”的数字
        For i = 0 To 9
            FeatureArr = Split(NumFeatures(i), "|")
            FirstColor = FeatureArr(0)
            OffsetStr = Replace(NumFeatures(i), FirstColor & "|", "")
            ' 在当前X坐标到区域右侧之间进行多点找色
            TempXY = Plugin.Color.FindMultiColor(CurrentX, y1, x2, y2, FirstColor, OffsetStr, Sim)
            ' 如果找到了，解析坐标
            If TempXY <> "-1|-1" Then
                ZB = InStr(TempXY, "|")
                FoundX = Clng(Left(TempXY, ZB - 1))
                ' 【核心修复】比较坐标，只保留X坐标最小的那个数字
                If FoundX < MinX Then
                    MinX = FoundX
                    MinNum = i
                End If

        ' 如果这一轮找到了最靠左的数字，则拼接到结果中，并移动起始坐标
        If MinNum <> -1 Then
            Result = Result & MinNum
            CurrentX = MinX + 4 ' 假设数字宽度约为15像素，向右移动
        Else
            Exit Do ' 如果没找到任何数字，说明这一行结束了，退出循环
        End If
    Loop
    If Result = "" Then
    即可落槌识别 = 0
Else
    即可落槌识别 = CLng(Result)
End If
End Function
