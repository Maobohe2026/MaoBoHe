Sub 向北寻路(目标X, 目标Y, 颜色, 偏色, 超时次数,间隔,相似度)
    寻路超时 = 0
    寻路出错 = 0
    X=0
    Do Until X=W+目标X and Y=H+目标Y
        Delay 间隔
        寻路超时 = 寻路超时 + 1
        If 寻路超时 > 超时次数 Then 
            KeyUp "W", 1
            KeyUp "S", 1
            KeyUp "A", 1
            KeyUp "D", 1
            Exit Sub
        End If
        XY=Plugin.Color.FindMultiColor(W+30,H+30,W+270,H+270,颜色,偏色,相似度)//寻路标记
        ZB = InStr(XY,"|")
        X = Clng(Left(XY, ZB - 1)) : Y = Clng(Right(XY, Len(XY) - ZB))
        TracePrint X-W&"-"&Y-H
        If X < 0 Then 
            Delay 300
            XY=Plugin.Color.FindMultiColor(W+30,H+30,W+270,H+270,颜色,偏色,相似度)//寻路标记
            ZB = InStr(XY,"|")
            X = Clng(Left(XY, ZB - 1)) : Y = Clng(Right(XY, Len(XY) - ZB))
            If X < 0 Then 
                Delay 300
                XY=Plugin.Color.FindMultiColor(W+30,H+30,W+270,H+270,颜色,偏色,相似度)//寻路标记
                ZB = InStr(XY,"|")
                X = Clng(Left(XY, ZB - 1)) : Y = Clng(Right(XY, Len(XY) - ZB))
                If X < 0 Then 
                    KeyUp "A", 1
                    Delay 100
                    KeyDown "D", 1
                    Delay 500
                    KeyUp "D", 1
                End If
            End If
        End If
        If XY = 上个XY Then 
            位置没变 = 位置没变 + 1
            If 位置没变 > 5 Then 
                If X < 0 Then 
                    Form1.ListBox1.AddItem "寻路找不到位置 退出进行下个路线"
                    Form1.ListBox1.ListIndex = Form1.ListBox1.ListCount - 1
                    寻路出错=1
                    Exit Sub
                End If
                KeyDown "A", 1
                Delay 130
                KeyUp "A", 1
                Delay 100
                KeyPress "X", 1
                KeyPress "Space", 2
            End If
        Else 
            位置没变 = 0
        End If
        上个XY=XY
        If 0 < X < W + 目标X - 3 Then 
            KeyUp "D", 1
            KeyDown "A", 1
        End If
        If W + 目标X-4 < X < W + 目标X Then 
            KeyDown "A", 1
            Delay 100
            KeyUp "A", 1
        End If
        If W+目标X+4 >X > W+目标X Then 
            KeyDown "D", 1
            Delay 100
            KeyUp "D", 1
        End If
        If X > W + 目标X + 3 Then 
            KeyUp "A", 1
            KeyDown "D", 1
        End If
        If X = W + 目标X  Then 
            KeyUp "D", 1
            KeyUp "A", 1
        End If
        If  Y = H + 目标Y Then 
            KeyUp "W", 1
            KeyUp "S", 1
        End If
        If H + 0 < Y < H + 目标Y - 2 Then 
            KeyUp "S", 1
            KeyDown "W", 1
        End If
        If H+目标Y-3<Y < H+目标Y Then 
            KeyDown "W", 1
            Delay 100
            KeyUp "W", 1
        End If
        If H+目标Y+4 > Y > H+目标Y Then

            Delay 100
            KeyUp "S", 1
        End If
        If Y > H + 目标Y + 3 Then 
            KeyUp "W", 1
            KeyDown "S", 1
        End If
    Loop

Sub 找方向
    TracePrint 1
    TracePrint 1
    TracePrint 1
    TracePrint 1
    TracePrint 1
    TracePrint 1
    TracePrint 1
    TracePrint 1
    Rem 重新找方向
    KeyPress "3", 2
    Form1.ListBox1.AddItem "寻找方向"
    Form1.ListBox1.ListIndex = Form1.ListBox1.ListCount - 1
    Dim Results(3, 1) ' 0列存X，1列存Y
    Dim Keys(3)
    Keys(0) = "W"
    Keys(1) = "S"
    Keys(2) = "A"
    Keys(3) = "D"
    Dim Found ' 定义一个标记，用于记录是否已经找到了绝对正确的方向
    Found = False
    ' 1. 依次按下四个方向键，每次按2秒后找色
    For i = 0 To 3
        KeyDown Keys(i), 1
        Delay 50 ' 按住 2 秒
        KeyUp Keys(i), 1
        Delay 900
        ' 找色
        XY = Plugin.Color.FindMultiColor(W+135, H+116, W+177, H+160, "B8EFFF", "1|1|B8EFFF", 1)
        ZB = InStr(XY, "|")
        TracePrint XY
        ' 解析并保存结果
        If ZB > 0 Then
            Results(i, 0) = Clng(Left(XY, ZB - 1)) ' 存入 X
            Results(i, 1) = Clng(Right(XY, Len(XY) - ZB)) ' 存入 Y
            ' 【核心逻辑】：如果 Y 值小于 280，直接确定方向！
            If Results(i, 1)-H < 131 Then 
                BestIndex=i
                TracePrint "提前确定按键：" & Keys(i) & "，Y值=" & Results(i, 1)
                Found = True
                Exit For ' 直接跳出循环，不再测试后面的按键
            End If
        Else
            Results(i, 0) = -1
            Results(i, 1) = 99999 ' 找色失败，给一个极大的Y值
        End If
    Next
    If Form1.TabControl1.Tab = 3 or Form1.CheckBox31.Value = 1 Then //锄地才用
                    XY=Plugin.Color.FindMultiColor(W+80,H+80,W+1860,H+1000,"BFFFFF","1|1|BFFFFF,2|2|BFFFFF,3|3|BFFFFF,4|4|BFFFFF,5|5|BFFFFF,6|6|BFFFFF",0.9)//锁定标记
                    ZB = InStr(XY,"|")
                    X = Clng(Left(XY, ZB - 1)) : Y = Clng(Right(XY, Len(XY) - ZB))
        If X > 0 or O > 0 Then 
            Form1.ListBox1.AddItem "竟然还有怪 继续战斗"
            Form1.ListBox1.ListIndex = Form1.ListBox1.ListCount - 1
            Call 战斗(4,4,0)
            Goto 重新找方向
        End If
    End If
    ' 2. 根据是否提前找到，决定后续逻辑
    If Found Then
        ' 提前找到了，直接使用当前的 i 作为最佳方向
        TracePrint "最终确定方向：" & Keys(i) & "，坐标：" & Results(i, 0) & "," & Results(i, 1)
        ' 这里可以执行后续操作，例如：MoveTo Results(i, 0), Results(i, 1)
    Else
        ' 4个方向都按完了，都没有 Y < 280 的情况，退而求其次找 Y 最小的
        MinY = 99999
        BestIndex = -1
        For j = 0 To 3
            If Results(j, 1) < MinY Then
                MinY = Results(j, 1)
                BestIndex = j
            End If
        Next
        If BestIndex = -1 Or MinY = 99999 Then
            TracePrint "4个方向均未找到目标！"
        Else
            TracePrint "选择相对最优方向：" & Keys(BestIndex) & "，坐标：X=" & MinX & ", Y=" & MinY
        End If
    End If
    KeyDown Keys(BestIndex), 1
    Delay 200 ' 按住 2 秒
    KeyUp Keys(BestIndex), 1
    Delay 1500
    KeyPress "Alt", 1
    Delay 100
    MiddleClick 1
    Delay 2000
    Form1.ListBox1.AddItem "确定大致方向 开始准确调整"
    Form1.ListBox1.ListIndex = Form1.ListBox1.ListCount - 1
    XY=Plugin.Color.FindMultiColor(W+135,H+116,W+177,H+160,"B8EFFF","1|1|B8EFFF",1)
    ZB = InStr(XY,"|")
    X = Clng(Left(XY, ZB - 1)) : Y = Clng(Right(XY, Len(XY) - ZB))
    If X-W < 153 Then 
        For 30
            XY=Plugin.Color.FindMultiColor(W+50,H+41,W+1883,H+1000,"2120F3","3|0|2120F3",0.9)
            ZB = InStr(XY,"|")
            X = Clng(Left(XY, ZB - 1)) : Y = Clng(Right(XY, Len(XY) - ZB))
            If X > 0 or O > 0 Then 
                Form1.ListBox1.AddItem "竟然还有怪 继续战斗"
                Form1.ListBox1.ListIndex = Form1.ListBox1.ListCount - 1
                Call 战斗(4,4,0)
